Events
| Event           | Description       |
| --------------- | ----------------- |
| `onClick`       | Mouse click       |
| `onDoubleClick` | Double click      |
| `onChange`      | Input change      |
| `onSubmit`      | Form submit       |
| `onMouseEnter`  | Mouse hover       |
| `onMouseLeave`  | Mouse out         |
| `onKeyDown`     | Key press         |
| `onFocus`       | Element focused   |
| `onBlur`        | Element unfocused |

  8)Add Item From Child → Show List in Parent
        import { useState } from "react";
        import AddItem from "./components/AddItem";
        export default function App() {
            const [items, setItems] = useState([]);
            const parentMessage = "Enter an item and click Add";

            function handleAddItem(newItem) {
              setItems([...items, newItem]);
            }

            return (
              <div >
                <AddItem instruction={parentMessage} onAdd={handleAddItem}/>
                <ul>
                  {items.map((item, i) => (<li key={i}>{item}</li>))}
                </ul>
              </div>
            );
        }

        import { useState } from "react";
        export default function AddItem({ instruction, onAdd }) {
            const [input, setInput] = useState("");

            function handleSubmit() {
              if (input.trim() === "") return alert("Please enter something");
              onAdd(input);      // sending data BACK TO PARENT
              setInput("");      // clear input
            }

            return (
              <div>
                <p>{instruction}</p>
                <input type="text" value={input} onChange={(e) => setInput(e.target.value)}/>
                <button onClick={handleSubmit}>Add</button>
              </div>
            );
          }

    to handle checkbox checked and update value
    import { useState } from "react";
      function App() {
        const [checkedValue, setCheckedValue] = useState(false);
        function handleChange(e) {
          setCheckedValue(e.target.checked);
        }
        return (
          <div>
            <input  type="checkbox" checked={checkedValue}  onChange={handleChange}/>
            <p>Checkbox is: {checkedValue ? "Checked" : "Not Checked"}</p>
          </div>
        );
      }
      export default App;
